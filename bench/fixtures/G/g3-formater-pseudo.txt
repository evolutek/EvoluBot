Code (outils/etiquettes.c) :

```c
/* Étiquette d'un casier : « pseudo [projet] », ou le pseudo seul si le membre n'a pas de projet
   (projet NULL ou chaîne vide). Le résultat est alloué, l'appelant le libère. */
char *formater_pseudo(const char *pseudo, const char *projet)
{
    size_t n = strlen(pseudo) + strlen(projet) + 3; /* " [" et "]" */
    char *s = malloc(n);
    if (!s)
        return NULL;
    sprintf(s, "%s [%s]", pseudo, projet);
    return s;
}
```
